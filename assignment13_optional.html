even_or_odd <- function(x) {

    • if the input vector has no missing value: return the same input vector

mean_impute <- function(x) {
  if (!is.numeric(x)) {
    stop("Input must be a numeric vector.")
  }
  if (any(is.na(x))) {
    if (all(is.na(x))) {
      stop("At least one observed value is needed to calculate a mean.")
    }
    x[is.na(x)] <- mean(x, na.rm = TRUE)
  }
  return(x)
}

mean_impute(c(10, 20, NA, 30))
## [1] 10 20 20 30
mean_impute(c(10, 20, 30))
## [1] 10 20 30

The first result is 10, 20, 20, 30. The missing entry is replaced by the mean of the observed values. The second input has no missing values and is returned unchanged. If all entries are missing, there is no observed mean, so the function gives an explanatory error.

impute_missing <- function(x) {
  if (any(is.na(x))) {
    if (all(is.na(x))) {
      stop("At least one observed value is needed for imputation.")
    }
    if (is.numeric(x)) {
      x <- mean_impute(x)
    } else {
      # Find the most frequent observed value, as in the slides.
      mode_of_x <- names(sort(table(x), decreasing = TRUE))[1]
      # Take that value from x to preserve its type, including factors.
      replacement <- x[match(mode_of_x, as.character(x))]
      x[is.na(x)] <- replacement
    }
  }
  return(x)
}

impute_missing(c(5, NA, 15))
## [1]  5 10 15
impute_missing(c("red", NA, "blue", "red"))
## [1] "red"  "red"  "blue" "red"
impute_missing(factor(c("yes", NA, "no", "yes")))
## [1] yes yes no  yes
## Levels: no yes
impute_missing(c(TRUE, NA, TRUE, FALSE))
## [1]  TRUE  TRUE  TRUE FALSE
impute_missing(c("red", "blue"))
## [1] "red"  "blue"

Numeric missing values are replaced by the observed mean. Non-numeric missing values are replaced by the mode, or most frequent observed value. The examples also show that factor and logical inputs keep their types. When modes tie, the first tied category in the table’s order is used. Inputs without missing values are returned unchanged.

boxplot_two <- function(d) {
  if (ncol(d) != 2) {
    stop("Provide a data frame with exactly two columns.")
  }
  if (is.numeric(d[[1]]) && !is.numeric(d[[2]])) {
    p <- ggplot(d, aes(x = d[[2]], y = d[[1]])) +
      geom_boxplot(fill = "steelblue") +
      labs(title = paste(names(d)[1], "by", names(d)[2]),

plot_two <- function(d) {
  if (ncol(d) != 2) {
    stop("Provide a data frame with exactly two columns.")
  }
  if (is.numeric(d[[1]]) && is.numeric(d[[2]])) {
    p <- ggplot(d, aes(x = d[[1]], y = d[[2]])) +
      geom_point(color = "steelblue", size = 2) +
      labs(title = paste(names(d)[2], "versus", names(d)[1]),
           x = names(d)[1], y = names(d)[2],
           caption = "Each point represents one row of the input data.")
    return(p)
  } else if (!is.numeric(d[[1]]) && !is.numeric(d[[2]])) {
    p <- ggplot(d, aes(x = d[[1]], fill = d[[2]])) +
      geom_bar(position = "dodge") +
      labs(title = paste(names(d)[1], "by", names(d)[2]),
           x = names(d)[1], y = "Count", fill = names(d)[2],
           caption = "Bars count rows in each combination of categories.")
    return(p)
  } else {
    return(boxplot_two(d))
  }
}

tree_accuracy <- function(input_data, train_percent = .8) {
  stopifnot("target" %in% names(input_data), !anyNA(input_data),
            train_percent > 0, train_percent < 1)
  input_data <- input_data %>%
    mutate(across(where(is.character), as.factor), target = factor(target))
  stopifnot(nlevels(input_data$target) == 2)

  set.seed(421)
  splitIndex <- createDataPartition(input_data$target, p = train_percent,
                                    list = FALSE)
  df_train <- input_data[splitIndex, ]
  df_test <- input_data[-splitIndex, ]

  tree_model <- rpart(target ~ ., data = df_train, method = "class")
  pred <- predict(tree_model, df_test, type = "class")
  cm <- confusionMatrix(data = pred, reference = df_test$target,
                         positive = levels(input_data$target)[2])
  return(unname(cm$overall["Accuracy"]))
}

  1. Write the following function. Give examples to test your function.

  • Input:

forest_importance <- function(input_data, train_percent = .8) {
  stopifnot("target" %in% names(input_data), !anyNA(input_data),
            train_percent > 0, train_percent < 1)
  input_data <- input_data %>%
    mutate(across(where(is.character), as.factor), target = factor(target))
  stopifnot(nlevels(input_data$target) == 2)

  set.seed(421)
  splitIndex <- createDataPartition(input_data$target, p = train_percent,
                                    list = FALSE)
  df_train <- input_data[splitIndex, ]

  ctrl <- trainControl(method = "cv", number = 5, allowParallel = FALSE)
  forest_model <- train(
    target ~ ., data = df_train, method = "ranger", trControl = ctrl,
    tuneGrid = expand.grid(mtry = c(1, 2), splitrule = "gini", min.node.size = 5),
    num.trees = 300, num.threads = 1, importance = "impurity"
  )
  importance_plot <- plot(varImp(forest_model, scale = FALSE),
                           main = "Random forest variable importance")
  return(importance_plot)
}

forest_importance(iris_binary)

forest_importance(iris_binary, train_percent = .7)

The function uses caret’s train() with method = "ranger" to fit a random forest. Five-fold cross-validation happens within the training portion. The returned plot ranks predictor importance for the fitted forest. Importance shows predictive use and does not establish causation.

text_wordcloud <- function(input_data, text_col) {
  stopifnot(text_col %in% names(input_data))
  text_data <- data.frame(text = as.character(input_data[[text_col]]))

  word_counts <- text_data %>%
    filter(!is.na(text)) %>%
    unnest_tokens(input = text, output = word) %>%
    anti_join(get_stopwords(), by = "word") %>%
    count(word, sort = TRUE)

  if (nrow(word_counts) == 0) {
    stop("There are no words to plot after removing missing text and stop words.")
  }
  set.seed(421)
  par(mar = c(1, 1, 4, 1))
  wordcloud(words = word_counts$word, freq = word_counts$n,
            min.freq = 1, max.words = 50, random.order = FALSE,
            scale = c(3, .7), colors = brewer.pal(8, "Dark2"))
  title(main = paste("Word cloud:", text_col))
}
